DOCUMENTO DE EJEMPLO CON LOS DATOS JSON DE TODOS LOS MÉTODOS POST

REALIZAR EN ESTE ORDEN PARA QUE LOS ID COINCIDAN


LOGUEAR COMO TERAPEUTA (POST) {USAR SOLO PARA TENER EL TOKEN DE INICIO DE TERAPEUTA} =
http://localhost:8080/login 
JSON
{
  "username": "admin",
  "password": "1234"
}


Crear paciente (POST) =
http://localhost:8080/paciente
JSON
{
  "nombre": "Juan",
  "apellido": "Pérez",
  "foto": "https://link-a-foto.com/juan.jpg",
  "correo": "juan.perez@example.com",
  "username": "paciente",
  "password": "1234",
  "rol": "PACIENTE",
  "consentimiento": true,
  "mayorDeEdad": true
}

Logear como paciente (POST) =
http://localhost:8080/login 
JSON
{
    "username": "paciente",
    "password": "1234"
}

CREAR CITA (POST) = 
http://localhost:8080/cita/1
JSON
{
  "fecha": "2023-10-01T12:00:00",  // Formato de fecha y hora
  "link": "https://www.ejemplo.com", // URL válida
  "anotacion": "Esta es una anotación de ejemplo.",
  "pagado": false,
  "tarea": "Completar el informe mensual"
}

CREAR INFORME (POST) =
http://localhost:8080/informe/3
JSON
{
    "demanda": "Prueba",
    "tarea": "Prueba"
}
